```yaml
    name: Test
    runs-on: ubuntu-latest

      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
      - uses: Swatinem/rust-cache@v2

    name: Clippy
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
        with:
          components: clippy
      - uses: Swatinem/rust-cache@v2
      - name: Run clippy
        run: cargo clippy -- -D warnings

  fmt:
    name: Format
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
        with:
          components: rustfmt
      - name: Check formatting
        run: cargo fmt --all -- --check
```

This workflow runs on every push to main/master and on PRs. It checks tests (with LOGTOK_KEY for store tests), clippy linting, and rustfmt formatting. Uses `Swatinem/rust-cache@v2` for dependency caching.

Create `.github/workflows/release.yml` (per D-16, D-17):

```yaml
name: Release

on:
  push:
    tags: ['v*']

permissions:
  contents: write

env:
  CARGO_TERM_COLOR: always

jobs:
  build:
    name: Build ${{ matrix.target }}
    strategy:
      fail-fast: false
      matrix:
        include:
          - target: x86_64-unknown-linux-musl
            os: ubuntu-latest
            use_cross: true
          - target: aarch64-unknown-linux-musl
            os: ubuntu-latest
            use_cross: true
          - target: x86_64-apple-darwin
            os: macos-latest
            use_cross: false
          - target: aarch64-apple-darwin
            os: macos-latest
            use_cross: false
          - target: x86_64-pc-windows-msvc
            os: windows-latest
            use_cross: false

    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4

      - uses: dtolnay/rust-toolchain@stable
        with:
          targets: ${{ matrix.target }}

      - uses: Swatinem/rust-cache@v2
        with:
          key: ${{ matrix.target }}

      - name: Install cross
        if: matrix.use_cross
        run: cargo install cross --locked

      - name: Build
        run: |
          if [ "${{ matrix.use_cross }}" = "true" ]; then
            cross build --release --target ${{ matrix.target }}
          else
            cargo build --release --target ${{ matrix.target }}
          fi
        shell: bash

      - name: Package (Unix)
        if: runner.os != 'Windows'
        run: |
          cd target/${{ matrix.target }}/release
          tar czf ../../../logtok-${{ matrix.target }}.tar.gz logtok
        shell: bash

      - name: Package (Windows)
        if: runner.os == 'Windows'
        run: |
          cd target/${{ matrix.target }}/release
          7z a ../../../logtok-${{ matrix.target }}.zip logtok.exe
        shell: bash

      - uses: actions/upload-artifact@v4
        with:
          name: logtok-${{ matrix.target }}
          path: logtok-${{ matrix.target }}.*

  release:
    name: Publish Release
    needs: build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/download-artifact@v4
        with:
          path: artifacts

      - name: Generate checksums
        run: |
          cd artifacts
          find . -name 'logtok-*' -type f | sort | while read f; do
            sha256sum "$f"
          done > ../checksums-sha256.txt
          cat ../checksums-sha256.txt

      - uses: softprops/action-gh-release@v2
        with:
          generate_release_notes: true
          files: |
            artifacts/logtok-*/logtok-*
            checksums-sha256.txt
```

Key design decisions:
- `fail-fast: false` so one platform failure doesn't cancel others
- `permissions: contents: write` required for creating releases
- macOS targets use `macos-latest` runner (not cross) per Pitfall 4 -- macOS SDK not available on Linux
- Linux targets use `cross` for musl static linking (no glibc dependency)
- Windows uses native `cargo build` on windows-latest runner
- Separate Package steps for Unix (tar.gz) and Windows (zip)
- `softprops/action-gh-release@v2` (latest major) with `generate_release_notes: true` for automatic changelog
- SHA256 checksums generated for all artifacts

| T-03-08 | Tampering | release.yml | mitigate | SHA256 checksums published alongside binaries. Users can verify integrity. Pin action versions (checkout@v4, rust-toolchain@stable, gh-release@v2) to prevent supply chain attacks via unpinned tags. |
