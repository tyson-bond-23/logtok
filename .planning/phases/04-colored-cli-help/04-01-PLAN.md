1. **test_no_color_compliance (HELP-02):** Verify NO_COLOR=1 strips ANSI codes from help output.
```rust
#[test]
fn test_no_color_compliance() {
    let output = Command::cargo_bin("logtok")
        .unwrap()
        .arg("--help")
        .env("NO_COLOR", "1")
        .output()
        .unwrap();
    let stdout = String::from_utf8_lossy(&output.stdout);
    // ANSI escape codes start with ESC[ (0x1b 0x5b)
    assert!(
        !stdout.contains("\x1b["),
        "Help output should not contain ANSI escape codes when NO_COLOR=1 is set. Found: {}",
        stdout
    );
}
```

2. **test_colored_output_contains_ansi (HELP-01):** Verify that without NO_COLOR, help output contains ANSI codes (when stdout is a pseudo-TTY -- we approximate by checking CLICOLOR_FORCE=1).
```rust
#[test]
fn test_colored_output_with_clicolor_force() {
    let output = Command::cargo_bin("logtok")
        .unwrap()
        .arg("--help")
        .env("CLICOLOR_FORCE", "1")
        .env_remove("NO_COLOR")
        .output()
        .unwrap();
    let stdout = String::from_utf8_lossy(&output.stdout);
    // With CLICOLOR_FORCE=1, ANSI codes should be present even when piped
    assert!(

        "Help output should contain ANSI escape codes when CLICOLOR_FORCE=1. Got: {}",

}
```
