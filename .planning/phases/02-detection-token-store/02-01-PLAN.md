       fn luhn_check(number: &str) -> bool {
           let digits: Vec<u32> = number.chars()
               .filter(|c| c.is_ascii_digit())
               .map(|c| c.to_digit(10).unwrap())
               .collect();
           if digits.len() < 13 || digits.len() > 19 { return false; }
           let sum: u32 = digits.iter().rev().enumerate().map(|(i, &d)| {

               else { d }
           }).sum();
           sum % 10 == 0
       }

       pub fn merge(&mut self, other: TokenMapData) {
           for (value, entry) in other.entries {
               if !self.data.value_to_token.contains_key(&value) {
                   self.data.value_to_token.insert(value.clone(), entry.token.clone());
                   self.data.token_to_value.insert(entry.token.clone(), value.clone());
                   self.data.entries.insert(value, entry.clone());
                   let counter = self.data.category_counters.entry(entry.category.clone()).or_insert(0);
                   // Parse counter from token format [CAT_NNN] and take max
                   if let Some(num) = parse_counter_from_token(&entry.token) {
                       if num > *counter { *counter = num; }
                   }
               }
           }
       }

       pub fn purge_expired(&mut self, ttl_seconds: u64) {
           let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
           let expired_values: Vec<String> = self.data.entries.iter()
               .filter(|(_, entry)| now - entry.created_at > ttl_seconds)
